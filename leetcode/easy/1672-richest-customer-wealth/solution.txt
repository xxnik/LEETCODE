class Solution {
public:
    int maximumWealth(vector<vector<int>>& accounts) {
        priority_queue<int> pq;
        for(int i=0;i<accounts.size();i++){
            int sum=0;
            for( int x:accounts[i]){
                sum+=x;
            }
            pq.push(sum);
        }
        return pq.top();
    }
};
