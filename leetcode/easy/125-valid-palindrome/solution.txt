class Solution {
public:
   
    bool checkpalindrome(const string &s,int i){
        int n=s.size();
        if(i>=n/2){
            return true ;
        }
        if(s[i]!=s[n-i-1]){return false;}
        return checkpalindrome(s,i+1);
        
    }
    bool isPalindrome(string s) {
        string p="";
        for(char c:s){
            if(isalnum(c)){
                p+=tolower(c);
            }
        }
        return checkpalindrome(p,0);
       

    }
};
