class Solution {
public:
    bool checkPalindrome(string &s, int left, int right) {
        if (left >= right) {
            return true;
        }

        if (s[left] != s[right]) {
            return false;
        }

        return checkPalindrome(s, left + 1, right - 1);
    }
    bool isPalindrome(string s) {
        string p="";
        for(char c:s){
            if(isalnum(c)){
                p+=tolower(c);
            }
        }int n=p.size();
        return checkPalindrome(p,0,n-1);
        
    }
};
