class Solution {
public:
    int lastStoneWeight(vector<int>& stones) {
        priority_queue<int> pq;
        for(int x:stones){
            pq.push(x);
        }
        while(pq.size()>=2){
            int a=pq.top();
            pq.pop();
            int b=pq.top();
            pq.pop();
            int res=a-b;
            pq.push(res);
        }
        return pq.top();
    }
};
