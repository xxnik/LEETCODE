class Solution {
public:
    void recursion(vector<char>&s,int l,int r){
        if(l>=r){
            return;
        }
        swap(s[l],s[r]);
        recursion(s,l+1,r-1);
    
    }
    void reverseString(vector<char>& s) {
        int n=s.size()-1;
        recursion(s,0,n);
        for (int i = 0; i < n; i++) {
            cout << s[i] << " ";
        }
        
    }
};
