nterval(vector<char>& tasks, int n) {
        unordered_map<char,int> mp;
        for(char c:tasks){
            mp[c]++;
        }
        int minInterval=INT_MAX;
        int count=0;
        int maxFreq=INT_MIN;
        for(auto it:mp){
           
            maxFreq=max(maxFreq,it.second);

        }
        for(auto it:mp){
            if(it.second==maxFreq){
                count++;
            }
        }
        int ans=(maxFreq-1)*(n+1)+count;
        return max((int)tasks.size(), ans);
    }
};
