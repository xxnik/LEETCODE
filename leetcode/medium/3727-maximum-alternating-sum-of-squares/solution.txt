class Solution {
public:
    long long maxAlternatingSum(vector<int>& nums) {
        
        vector<int> sqr;
        for(int x:nums){
            sqr.push_back(x*x);
        }
        sort(sqr.begin(),sqr.end());
        int n=sqr.size();
        long long sum=0;
        for(int i=0;i<n/2;i++){
            int a=sqr[i];
            sum=sum-a;
        }
        for(int i=n/2;i<n;i++){
            int b=sqr[i];
            sum=sum+b;
        }
        return sum;
    }
};
