lass Solution {
public:
    bool canSortArray(vector<int>& nums) {
        while(true){
            bool swapped=false;
            for(int i=0;i<nums.size()-1;i++){
                int j=i+1;
                if(nums[i]>nums[j]){
                    if(__builtin_popcount(nums[i])==__builtin_popcount(nums[j])){
                        swap(nums[i],nums[j]);
                        swapped=true;
                    }
                }
            
            }
            if(swapped==false)break;
        }
        return is_sorted(nums.begin(),nums.end())?true:false;
    }
};
