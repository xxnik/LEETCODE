class Solution {
public:

    void recsum(int i,int n,int target,vector<int>& candidates,vector<int>& ans,vector<vector<int>>& ds){
        if(target==0){
            ds.push_back(ans);
            return;
        }
        if(i>=n||target<0){
            return;
        }

        ans.push_back(candidates[i]);
        recsum(i+1,n,target-candidates[i],candidates,ans,ds);
        ans.pop_back();
        int next=i+1;
        while(next<n&&candidates[next]==candidates[i]){
            next++;
        }
        recsum(next,n,target,candidates,ans,ds);
    }
    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        vector<vector<int>> ds;
        vector<int> ans;
        sort(candidates.begin(), candidates.end());
        int i=0;
        int n=candidates.size();
        recsum(i,n,target,candidates,ans,ds);

        return ds;

    }
};
