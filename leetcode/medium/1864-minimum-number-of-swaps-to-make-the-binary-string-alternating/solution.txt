0;
        for(int x:s){
            if(x=='1'){
                cnt++;
            }
        }
        return cnt;
    }
    int minSwaps(string s) {
        int n=s.size();
        int cnt1=count(s);
        int cnt0=n-cnt1;
        if(abs(cnt1-cnt0)>1)return -1;
        int m0=0;
        int m1=0;
        for(int i=0;i<n;i++){
           if(i%2==0){
            if(s[i]=='1'){
                m0++;
            }else{
                m1++;
            }
           }else{
            if(s[i]=='0'){
                m0++;
            }else{
                m1++;
            }
           }
        }
        if(cnt0>cnt1)return m0/2;
        if(cnt1>cnt0)return m1/2;

        return min(m0,m1)/2;
        
    }
