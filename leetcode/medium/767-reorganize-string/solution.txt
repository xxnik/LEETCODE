t : freq) {
            pq.push({it.second, it.first});
        }

        string ans = "";

        while (pq.size() >= 2) {
            auto [freq1, ch1] = pq.top();
            pq.pop();

            auto [freq2, ch2] = pq.top();
            pq.pop();

            ans += ch1;
            ans += ch2;

            freq1--;
            freq2--;

            if (freq1 > 0)
                pq.push({freq1, ch1});

            if (freq2 > 0)
                pq.push({freq2, ch2});
        }

        
        if (!pq.empty()) {
            auto [f, ch] = pq.top();

            if (f > 1)
                return "";

            ans += ch;
        }

        return ans;
    }
