class Solution {
public:
    vector<int> findClosestElements(vector<int>& arr, int k, int x) {
        priority_queue<pair<int,int>> minH;
        vector<int> res;
        for(int i=0;i<arr.size();i++){
            minH.push({abs(arr[i]-x),arr[i]});
            if(minH.size()>k){
                minH.pop();
            }
        }
        while(minH.size()>0){
            res.push_back(minH.top().second);
            minH.pop();
        }
        sort(res.begin(),res.end());
        return res;
    }
};
